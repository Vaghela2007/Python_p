Program 6: Quntityerror
==================================================
Code:
==================================================
class PriceError(Exception):
    pass
class QuantityError(Exception):
    pass
try:
    price = int(input("Enter product price: "))
    quantity = int(input("Enter product quantity: "))
    if not(price < 100 and price<1000):
        raise PriceError("Price must be 1 digit")
    elif quantity <= 0:
        raise QuantityError("Quantity must be greater than 0")
    else:
        print("Product price and quantity are valid.")
        print("Price:", price)
        print("Quantity:", quantity)
except PriceError as e:
    print("Price Error:", e)
except QuantityError as e:
    print("Quantity Error:", e)
finally:
    print("Success: Program execution completed.")
Output:
==================================================
Product price and quantity are valid.
Price: 50
Quantity: 2
Success: Program execution completed.

class Employee:
    def __init__(self, eid=0, enm="", bs=0):
        self.eid = eid
        self.enm = enm
        self.bs = bs
        self.hra = 0
        self.da = 0
        self.pf = 0
        self.ns = 0

    def getData(self):
        self.eid = int(input("Enter employee ID: "))
        self.enm = input("Enter employee name: ")
        self.bs = int(input("Enter employee salary: "))
        self.calSalary()

    def calSalary(self):
        if self.bs < 5000:
            self.hra = self.bs * 0.10
            self.da = self.bs * 0.20
            self.pf = self.bs * 0.05

        elif self.bs < 10000:
            self.hra = self.bs * 0.20
            self.da = self.bs * 0.40
            self.pf = self.bs * 0.10

        else:
            self.hra = self.bs * 0.30
            self.da = self.bs * 0.60
            self.pf = self.bs * 0.15

        self.ns = self.bs + self.hra + self.da - self.pf

    def putData(self):
        print("Employee ID   :", self.eid)
        print("Employee Name :", self.enm)
        print("Basic Salary  :", self.bs)
        print("HRA           :", self.hra)
        print("DA            :", self.da)
        print("PF            :", self.pf)
        print("Net Salary    :", self.ns)
        print("--------------------------------")
